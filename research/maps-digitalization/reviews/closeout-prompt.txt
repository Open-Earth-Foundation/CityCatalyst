def rings(items):
    result=[];current=[]
    for it in items:
        if it[0]=='re':
            r=it[1];pts=[(r.x0,r.y0),(r.x1,r.y0),(r.x1,r.y1),(r.x0,r.y1)]
            result.append(pts if it[2]>0 else pts[::-1]);continue
        if it[0]!='l':raise ValueError('Unsupported curved path: require explicit curve tolerance before exporting')
        a,b=tuple(it[1]),tuple(it[2])
        if current and current[-1]!=a:result.append(current);current=[]
        if not current:current=[a]
        current.append(b)
    if current:result.append(current)
    return [r if r[0]==r[-1] else r+[r[0]] for r in result if len(set(r))>=3]

def parts(g):
    if g.is_empty:return []
    if g.geom_type=='Polygon':return [g]
    return [p for c in getattr(g,'geoms',[]) for p in parts(c)]

def boundary_distances(a,b):
    """Vertex-to-segment distances in both directions, indexed rather than quadratic."""
    def one(source,target):
        segments=[]
        for p in parts(target):
            for ring in [p.exterior,*p.interiors]:
                c=np.asarray(ring.coords);segments.append(np.stack([c[:-1],c[1:]],axis=1))
        tree=shapely.STRtree(shapely.linestrings(np.concatenate(segments)))
        points=shapely.points(shapely.get_coordinates(source.boundary))
        _,distance=tree.query_nearest(points,return_distance=True,all_matches=False)
        return distance
    d=np.r_[one(a,b),one(b,a)]*7.62
    return {'median_m':float(np.median(d)),'p95_m':float(np.percentile(d,95)),'p99_9_m':float(np.percentile(d,99.9)),'max_m':float(d.max()),
      'vertices_over_0_1m':int((d>.1).sum()),'vertices_checked':len(d)}

def fill_geometry(d):
    rr=rings(d['items'])
    polygons=[Polygon(r) for r in rr]
    if d['even_odd'] and all(p.is_valid for p in polygons):
        gs=polygons
        while len(gs)>1:gs=[gs[i].symmetric_difference(gs[i+1]) if i+1<len(gs) else gs[i] for i in range(0,len(gs),2)]
        return gs[0] if gs else Polygon()
    # Crossing parity also handles self-intersecting or multiply-wound even-odd rings.
    faces=polygonize(unary_union([LineString(r) for r in rr]));kept=[]
    if not rr:return Polygon()
    aa=np.concatenate([np.asarray(r[:-1]) for r in rr]);bb=np.concatenate([np.asarray(r[1:]) for r in rr])
    ax,ay=aa.T;bx,by=bb.T
    for face in faces:
        p=face.representative_point();px,py=p.x,p.y
        cross=(bx-ax)*(py-ay)-(px-ax)*(by-ay)
        w=int(np.sum((ay<=py)&(py<by)&(cross>0)))-int(np.sum((by<=py)&(py<ay)&(cross<0)))
        if (w%2!=0 if d['even_odd'] else w!=0):kept.append(face)
    return unary_union(kept)

    gx=np.floor((aff['mx']*xx+aff['bx']-E0)/CELL).astype(int)
    gy=np.floor((N1-aff['my']*yy-aff['by'])/CELL).astype(int)
    ok=(gx>=0)&(gx<NX)&(gy>=0)
    return yy[ok],xx[ok],gy[ok]*NX+gx[ok]

def classify_nearest(pixels):
    pixels=np.asarray(pixels,dtype=np.float32)
    d=((pixels[:,None,:]-RAMP[None,:,:])**2).sum(2)
    dn=((pixels[:,None,:]-NONDATA[None,:,:])**2).sum(2).min(1)
    steps=d.argmin(1);best=d.min(1)
    valid=(best<=42**2)&(best<dn)
    return np.where(valid,steps,-1),np.sqrt(best)

def classify_blended(pixels,min_alpha):
    """Fit a palette swatch composited with neutral map furniture; never infer through masks."""
    pixels=np.asarray(pixels,dtype=np.float32)
    best=np.full(len(pixels),np.inf,dtype=np.float32);labels=np.full(len(pixels),-1,dtype=np.int16)
    for k,swatch in enumerate(RAMP):
        for bg in (90.,128.,180.,226.,255.):
            direction=swatch-bg
            alpha=np.clip(((pixels-bg)*direction).sum(1)/(direction*direction).sum(),0,1)
            error=np.sqrt(((pixels-(bg+alpha[:,None]*direction))**2).sum(1))
            cost=error+2*(1-alpha)
            take=(alpha>=min_alpha)&(cost<best)
            best[take]=cost[take];labels[take]=k
    dn=np.sqrt(((pixels[:,None,:]-NONDATA[None,:,:])**2).sum(2).min(1))
    # Dark legend reds have blue above green (maximum 13); retain those exact swatches.
    red=(pixels[:,0]>pixels[:,1]+2)&(pixels[:,0]>pixels[:,2]+4)&(pixels[:,1]>=pixels[:,2]-16)
    valid=(best<=14)&(best<dn)&red
    return np.where(valid,labels,-1),best

def aggregate(ids,steps,support=3,median=False):
    if len(ids)==0 or len(ids)!=len(steps) or support<1:raise ValueError('Invalid aggregation inputs')
    keep=steps>=0;k=ids[keep];v=(steps[keep].astype(float)+.5)/10
    counts=np.bincount(k,minlength=int(ids.max())+1);totals=np.bincount(ids,minlength=len(counts))
    sums=np.bincount(k,weights=v,minlength=len(counts));good=np.flatnonzero(counts>=support)
    means=np.divide(sums,counts,out=np.zeros(len(counts)),where=counts>0)
    if median:
        order=np.argsort(k);ks=k[order];vs=v[order]
        for idx,group in zip(*np.unique(ks,return_index=True)):
            means[idx]=np.median(vs[group:group+counts[idx]])
    vals={int(i):float(means[i]) for i in good}
    meta={int(i):{'recognized_pixels':int(counts[i]),'municipal_pixels':int(totals[i]),'support_fraction':float(counts[i]/totals[i]),
                 'bin_quantization_half_width':.05,'status':'estimated_from_observed_colours'} for i in good}
    return vals,meta

def polygon_for_id(k):
    y,x=divmod(int(k),NX);e=E0+x*CELL;n=N1-y*CELL
    ring=[[round(z,7) for z in INV.transform(ee,nn)] for ee,nn in [(e,n),(e,n-CELL),(e+CELL,n-CELL),(e+CELL,n),(e,n)]]
    return {'type':'Polygon','coordinates':[ring]}

def export_cells(path,hazard,values,support,provenance):
    from shapely.geometry import shape
    features=[]
    for k,v in sorted(values.items()):
        if not np.isfinite(v) or not 0<=v<=1:raise ValueError('Risk value must be finite in [0,1]')
        geom=polygon_for_id(k)
        if not shape(geom).is_valid:raise ValueError('Final cell geometry invalid')
        features.append({'type':'Feature','id':k,'properties':{'hazard':hazard,'value':v,'class_index':min(int(np.floor(v*5)),4),**support.get(k,{})},'geometry':geom})
    write_json(path,{'type':'FeatureCollection','provenance':provenance,'features':features})

import json,hashlib

def reference(hazard,min_pixels=15):
    manifest=json.loads((INPUT/'arvc-manifest.json').read_text(encoding='utf-8'))
    if 'EPSG:3857' not in manifest['provenance']['encoding']['format']:raise ValueError('Unsupported raster encoding CRS')
    entry=manifest['layers']['arvc_off_risk_'+hazard+'_2050']
    im=Image.open(INPUT/('arvc_off_risk_'+hazard+'_2050.png'))
    transparency=im.info.get('transparency')
    valid_alpha=transparency==255 or (isinstance(transparency,bytes) and len(transparency)==256 and transparency[:255]==bytes([255])*255 and transparency[255]==0)
    if im.mode!='P' or not valid_alpha:raise ValueError('Unsupported indexed PNG / no-data encoding')
    idx=np.asarray(im);y,x=np.where(idx!=255)
    (south,west),(north,east)=entry['bounds'];proj=Transformer.from_crs(4326,3857,always_xy=True)
    xmin,ymin=proj.transform(west,south);xmax,ymax=proj.transform(east,north)
    e,n=Transformer.from_crs(3857,31982,always_xy=True).transform(xmin+(x+.5)*(xmax-xmin)/im.width,ymax-(y+.5)*(ymax-ymin)/im.height)
    gx=np.floor((e-E0)/CELL).astype(int);gy=np.floor((N1-n)/CELL).astype(int)
    assert np.all((gx>=0)&(gx<NX)&(gy>=0))
    ids=gy*NX+gx;counts=np.bincount(ids);sums=np.bincount(ids,weights=idx[y,x]/254)
    return {int(k):float(sums[k]/counts[k]) for k in np.flatnonzero(counts>=min_pixels)},counts

def split(k):
    y,x=divmod(int(k),NX)
    fold=int.from_bytes(hashlib.sha256(f'{x//12}:{y//12}'.encode()).digest()[:4],'big')%5
    return 'test' if fold==0 else 'validation' if fold==1 else 'development'

def metrics(reference_values,counts,prediction,subset=None,jvp=False):
    ref={k:v for k,v in reference_values.items() if subset is None or split(k)==subset}
    pred={k:v for k,v in prediction.items() if subset is None or split(k)==subset}
    common=sorted(ref.keys()&pred.keys())
    if not common or not ref:raise ValueError('Empty comparison domain')
    a=np.array([ref[k] for k in common]);b=np.array([pred[k] for k in common])
    ca=np.minimum(np.floor(a*5).astype(int),4)
    cb=np.minimum(np.rint(b*9).astype(int)//2,4) if jvp else np.minimum(np.floor(b*5).astype(int),4)
    correct=int((ca==cb).sum());high_total=sum(v>=.6 for v in ref.values())
    zero_reference=[k for k in pred if k>=len(counts) or counts[k]==0]
    low_reference=[k for k in pred if k<len(counts) and 0<counts[k]<15]
    predicted_high=int((cb>=3).sum())
    return {'eligible':len(ref),'common':len(common),'missing':len(ref)-len(common),'missing_pct':100*(len(ref)-len(common))/len(ref),
        'exact_common_pct':100*correct/len(common),'exact_all_pct':100*correct/len(ref),'bias':float((b-a).mean()),'mae':float(abs(b-a).mean()),
        'high_recall_pct':100*int(((ca>=3)&(cb>=3)).sum())/high_total if high_total else None,
        'high_precision_common_pct':100*int(((ca>=3)&(cb>=3)).sum())/predicted_high if predicted_high else None,
        'predicted_low_reference_pct':100*len(low_reference)/max(len(pred),1),
        'predicted_cells':len(pred),'predicted_zero_reference_cells':len(zero_reference),'predicted_zero_reference_pct':100*len(zero_reference)/max(len(pred),1)}

# A better recovery route for JVP's two examples

## What was achieved

**Richfield has a substantially better source reconstruction:** five valid vector layers extracted directly from the PDF, including fill-and-stroke paths, with placement from the document's embedded GeoPDF coordinates. This avoids both pixel tracing and fitting placement to an approximate city boundary.

**Porto has a better source-data route and a promising but rejected PDF trial.** Six numerical-source layers are ingested from the published indexed raster derivatives. Six scale-corrected existing layers retain the original coverage. A separate colour-mixing trial improves class recovery but fails coverage/precision gates, so all six trial layers remain research-only.

No generative model supplies geometry or hazard values. Local ML was not added: the important Richfield information was already in the file, while Porto's failure is ambiguous colour mixing and unsupported coverage. A larger model would not make an unknown masked pixel an observation.

Per-lake errors and unmatched candidates are in `outputs/richfield/validation.json`.

## Which Porto output to use

1. **Numerical-source route:** ingest the published indexed PNGs directly (`index / 254`, 255 = no-data), retain all cells with at least one source pixel and flag cells below 15. The six source rasters and manifest are preserved for finer detail. Exact readback tests validate ingestion only. The original TIFFs have not been independently authenticated.
2. **Existing-PDF fallback:** apply `0.9 * v + 0.05` to repair the numeric bin-centre scale. Intended classes and missing counts remain unchanged; bias/MAE improve. Missing means unknown, not source-masked or zero risk.
3. **Experimental PDF recovery:** useful for research and visually inspecting candidate missing regions. It is not an approved substitute for the source data. Palette intervals exclude classification error and are not confidence intervals.
