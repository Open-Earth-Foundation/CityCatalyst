def rings(items):

def parts(g):

    return [p for c in getattr(g,'geoms',[]) for p in parts(c)]

def fill_geometry(d):
    rr=rings(d['items'])
    if d['even_odd']:
        gs=[unary_union(parts(shapely.make_valid(Polygon(r)))) for r in rr]
        while len(gs)>1:gs=[gs[i].symmetric_difference(gs[i+1]) if i+1<len(gs) else gs[i] for i in range(0,len(gs),2)]
        return gs[0] if gs else Polygon()
    # Nonzero winding is needed for clipping; preserve orientation rather than XOR.
    faces=polygonize(unary_union([LineString(r) for r in rr]));kept=[]
    for face in faces:
        p=face.representative_point();w=0
        for r in rr:
            for (ax,ay),(bx,by) in zip(r,r[1:]):
                cross=(bx-ax)*(p.y-ay)-(p.x-ax)*(by-ay)
                if ay<=p.y<by and cross>0:w+=1
                if by<=p.y<ay and cross<0:w-=1
        if w:kept.append(face)
    return unary_union(kept)

def source_shapes(page,colours,layer_predicate):
    stack=[];cache={};found=[];counts=Counter()
    for d in page.get_drawings(extended=True):

        g=fill_geometry(d)

    return classes,dict(counts),float(overlap)

def mask_geometry(g,w,h,scale):

    shapely.prepare(g);mask=np.zeros((h,w),bool);xs=(np.arange(w)+.5)/scale

        mask[start:start+len(ys)]=shapely.contains_xy(g,xs[None,:],ys[:,None])
    return mask

def reference(hazard,min_pixels=15):
    manifest=json.loads((INPUT/'arvc-manifest.json').read_text(encoding='utf-8'))
    entry=manifest['layers']['arvc_off_risk_'+hazard+'_2050']
    im=Image.open(INPUT/('arvc_off_risk_'+hazard+'_2050.png'))
    assert im.mode=='P' and im.info['transparency']==255
    idx=np.asarray(im);y,x=np.where(idx!=255)
    (south,west),(north,east)=entry['bounds'];proj=Transformer.from_crs(4326,3857,always_xy=True)
    xmin,ymin=proj.transform(west,south);xmax,ymax=proj.transform(east,north)
    e,n=Transformer.from_crs(3857,31982,always_xy=True).transform(xmin+(x+.5)*(xmax-xmin)/im.width,ymax-(y+.5)*(ymax-ymin)/im.height)
    gx=np.floor((e-E0)/CELL).astype(int);gy=np.floor((N1-n)/CELL).astype(int)
    assert np.all((gx>=0)&(gx<NX)&(gy>=0))
    ids=gy*NX+gx;counts=np.bincount(ids);sums=np.bincount(ids,weights=idx[y,x]/254)
    return {int(k):float(sums[k]/counts[k]) for k in np.flatnonzero(counts>=min_pixels)},counts

def split(k):
    y,x=divmod(int(k),NX)
    fold=int.from_bytes(hashlib.sha256(f'{x//12}:{y//12}'.encode()).digest()[:4],'big')%5
    return 'test' if fold==0 else 'validation' if fold==1 else 'development'

def metrics(reference_values,counts,prediction,subset=None,jvp=False):
    ref={k:v for k,v in reference_values.items() if subset is None or split(k)==subset}
    pred={k:v for k,v in prediction.items() if subset is None or split(k)==subset}
    common=sorted(ref.keys()&pred.keys())
    if not common or not ref:raise ValueError('Empty comparison domain')
    a=np.array([ref[k] for k in common]);b=np.array([pred[k] for k in common])
    ca=np.minimum(np.floor(a*5).astype(int),4)
    cb=np.minimum(np.rint(b*9).astype(int)//2,4) if jvp else np.minimum(np.floor(b*5).astype(int),4)
    correct=int((ca==cb).sum());high_total=sum(v>=.6 for v in ref.values())
    zero_reference=[k for k in pred if k>=len(counts) or counts[k]==0]
    low_reference=[k for k in pred if k<len(counts) and 0<counts[k]<15]
    predicted_high=int((cb>=3).sum())
    return {'eligible':len(ref),'common':len(common),'missing':len(ref)-len(common),'missing_pct':100*(len(ref)-len(common))/len(ref),
        'exact_common_pct':100*correct/len(common),'exact_all_pct':100*correct/len(ref),'bias':float((b-a).mean()),'mae':float(abs(b-a).mean()),
        'high_recall_pct':100*int(((ca>=3)&(cb>=3)).sum())/high_total if high_total else None,
        'high_precision_common_pct':100*int(((ca>=3)&(cb>=3)).sum())/predicted_high if predicted_high else None,
        'predicted_low_reference_pct':100*len(low_reference)/max(len(pred),1),
        'predicted_cells':len(pred),'predicted_zero_reference_cells':len(zero_reference),'predicted_zero_reference_pct':100*len(zero_reference)/max(len(pred),1)}

"""Run the two document profiles; use --route to avoid regenerating unrelated artifacts."""
import argparse,json,hashlib,importlib.metadata
import numpy as np
from PIL import features
from porto import *
from benchmark import reference,metrics

def porto():
    baseline=json.loads((INPUT/'arvc-poa.json').read_text(encoding='utf-8'))
    selection=json.loads((ROOT/'SELECTION.json').read_text(encoding='utf-8'))
    report={'selection_sha256':digest(ROOT/'SELECTION.json'),'hazards':{},'reference_warning':'Same-author indexed PNG derivatives. Original TIFF provenance was not authenticated. Native ingestion is not scored against itself.'}
    for hazard,page in PAGES.items():
        rgb=figure(page);aff=fit_ticks(rgb);yy,xx,ids=map_domain(rgb,aff)
        labels,_=classify_blended(rgb[yy,xx].astype(np.float32),selection['configuration']['minimum_alpha'])
        vals,support=aggregate(ids,labels,selection['configuration']['recognized_pixel_minimum'])
        # A bin interval excludes classification error; explicitly avoid a confidence claim.
        for k,v in vals.items():
            support[k].update(value_lower=max(0,v-.05),value_upper=min(1,v+.05),interval_kind='palette quantization only, NOT confidence or classification uncertainty',release_status='research_only_failed_coverage_gate')
        prov={'method':'PDF colour mixing experiment; failed validation coverage gate; not recommended for decisions','source_pdf_sha256':digest(INPUT/'porto-alegre-arvc.pdf'),
          'page':page,'decoded_rgb_sha256':hashlib.sha256(rgb.tobytes()).hexdigest(),'decoder':'Pillow '+importlib.metadata.version('pillow')+'; libjpeg '+str(features.version_codec('jpg')),
          'affine':aff,'selection_sha256':report['selection_sha256'],'cell_geometry':'250m aggregation square, not a traced risk footprint'}
        export_cells(OUT/'porto-experimental'/(hazard+'.geojson'),hazard,vals,support,prov)
        totals=np.bincount(ids);recognized=np.bincount(ids[labels>=0],minlength=len(totals))
        write_json(OUT/'porto-experimental'/(hazard+'-missing.json'),{'note':'Absence means insufficient observed colour support; it does not identify source masking.',
          'cells':[{'id':int(k),'reason':'no_accepted_colour' if recognized[k]==0 else 'insufficient_accepted_pixels','accepted_pixels':int(recognized[k]),'municipal_pixels':int(totals[k])} for k in np.flatnonzero((totals>0)&(recognized<8))]})
        raw={int(k):v for k,v in baseline['hazards'][hazard+'_risk_2050'].items()}
        corrected={k:float(.9*v+.05) for k,v in raw.items()}
        correction_meta={k:{'status':'existing_reconstruction_scale_corrected','value_lower':max(0,v-.05045),'value_upper':min(1,v+.05045),'interval_kind':'palette plus stored rounding only; classification/coverage uncertainty excluded'} for k,v in corrected.items()}
        export_cells(OUT/'porto-corrected-existing'/(hazard+'.geojson'),hazard,corrected,correction_meta,{'method':'Existing JVP reconstruction migrated to printed 0.1-bin midpoint scale; NOT new source recovery','input_sha256':digest(INPUT/'arvc-poa.json'),'missing_means':'Unknown: may be source mask OR extraction omission; never assume zero risk','cell_geometry':'250m aggregation square, not a risk footprint'})
        # Native route preserves every cell with >=1 numerical source pixel, including low-support edges.
        native,counts=reference(hazard,min_pixels=1)
        metadata={k:{'source_valid_pixels':int(counts[k]),'low_support':bool(counts[k]<15),'status':'aggregated_published_raster_derivative'} for k in native}
        target=OUT/'porto-raster-source'/(hazard+'.geojson')
        export_cells(target,hazard,native,metadata,{'method':'Numerical ingestion of published indexed PNG; value=index/254, 255=no-data, unweighted valid-pixel-centre mean in 250m cells','input_sha256':digest(INPUT/('arvc_off_risk_'+hazard+'_2050.png')),'manifest_sha256':digest(INPUT/'arvc-manifest.json'),'original_TIFF_authenticated':False,'cell_geometry':'250m aggregation square; preserve source PNG for ~29m detail; low-support edge cells flagged'})
        loaded=json.loads(target.read_text(encoding='utf-8'));assert {int(f['id']):f['properties']['value'] for f in loaded['features']}==native
        ref,_=reference(hazard)
        comparisons={}
        for part in ['validation','test',None]:
            comparisons[part or 'all_descriptive']={'jvp':metrics(ref,counts,raw,part,jvp=True),'scale_fix_only':metrics(ref,counts,corrected,part),'experimental':metrics(ref,counts,vals,part)}
        report['hazards'][hazard]={'comparisons':comparisons,'native_ingestion':{'cells':len(native),'low_support_cells':sum(counts[k]<15 for k in native).__int__(),'readback_values_identical':True},
          'eligibility_sensitivity_test':{str(n):metrics(reference(hazard,n)[0],counts,vals,'test') for n in [5,30]}}
        print(hazard,'test',json.dumps(comparisons['test']),flush=True)
    names=['jvp','scale_fix_only','experimental']
    report['test_macro_average']={name:{key:float(np.mean([report['hazards'][h]['comparisons']['test'][name][key] for h in PAGES])) for key in ['exact_all_pct','exact_common_pct','missing_pct','mae','bias','predicted_zero_reference_pct']} for name in names}
    write_json(OUT/'porto-benchmark.json',report)
    write_json(OUT/'runtime.json',{'versions':{p:importlib.metadata.version(p) for p in ['numpy','scipy','pillow','shapely','pyproj','pymupdf']},'input_hashes':{p.name:digest(p) for p in INPUT.iterdir() if p.is_file()},'OpenRouter_spent_USD':0})
    print(json.dumps(report['test_macro_average'],indent=2),flush=True)

import unittest,json,tempfile

def path(rings,evenodd=True,**kw):
    items=[]
    for ring in rings:
        for a,b in zip(ring,ring[1:]+ring[:1]):items.append(('l',pymupdf.Point(*a),pymupdf.Point(*b)))
    return {'items':items,'even_odd':evenodd,**kw}

class RecoveryChecks(unittest.TestCase):
    def test_fill_rules_and_holes(self):
        outer=[(0,0),(10,0),(10,10),(0,10)];inner=[(2,2),(8,2),(8,8),(2,8)]
        self.assertAlmostEqual(r.fill_geometry(path([outer,inner])).area,64)
        self.assertAlmostEqual(r.fill_geometry(path([outer,inner],False)).area,100)
        self.assertAlmostEqual(r.fill_geometry(path([outer,inner[::-1]],False)).area,64)
    def test_clipping_fill_and_stroke_and_draw_order(self):
        clip=path([[(0,0),(8,0),(8,8),(0,8)]],False,type='clip',level=0)
        a=path([[(0,0),(10,0),(10,10),(0,10)]],type='fs',level=1,fill=(1,0,0),fill_opacity=1,layer='data',seqno=0)
        b=path([[(4,0),(10,0),(10,10),(4,10)]],type='f',level=1,fill=(0,0,1),fill_opacity=1,layer='data',seqno=1)
        class Page:
            def get_drawings(self,**kwargs):return [clip,a,b]
        cs,counts,overlap=r.source_shapes(Page(),{'a':(255,0,0),'b':(0,0,255)},lambda s:s=='data')
        self.assertEqual(counts,{'fs':1,'f':1});self.assertEqual(cs['a'].area,32);self.assertEqual(cs['b'].area,32);self.assertEqual(overlap,32)
    def test_curve_fails_closed(self):
        with self.assertRaises(ValueError):r.rings([('c',)])
    def test_support_missing_and_midpoints(self):
        ids=np.array([0,0,0,1,1,2]);steps=np.array([0,0,-1,9,-1,-1])
        vals,meta=p.aggregate(ids,steps,support=2)
        self.assertEqual(set(vals),{0});self.assertAlmostEqual(vals[0],.05);self.assertEqual(meta[0]['recognized_pixels'],2)
    def test_uint8_colour_input(self):
        labels,_=p.classify_nearest(p.RAMP.astype(np.uint8));self.assertEqual(labels.tolist(),list(range(10)))
    def test_missing_denominator_and_zero_reference(self):
        m=metrics({0:.4,1:.7},np.array([20,20,0]),{0:.4,2:.5})
        self.assertEqual(m['exact_all_pct'],50);self.assertEqual(m['exact_common_pct'],100);self.assertEqual(m['missing_pct'],50);self.assertEqual(m['predicted_zero_reference_cells'],1)
    def test_cell_geometry_and_input_validation(self):
        g=shape(p.polygon_for_id(21547));self.assertTrue(g.is_valid);self.assertTrue(g.exterior.is_ccw)
        with tempfile.TemporaryDirectory() as t:
            with self.assertRaises(ValueError):p.export_cells(Path(t)/'x.json','heat',{0:float('nan')},{},{})
    def test_spatial_split_shared_block(self):
        self.assertEqual(split(0),split(11+11*p.NX))
    def test_all_delivered_geometry(self):
        paths=list(p.OUT.glob('*/*.geojson'));self.assertGreaterEqual(len(paths),23)
        for file in paths:
            data=json.loads(file.read_text(encoding='utf-8'))
            for f in data['features']:self.assertTrue(shape(f['geometry']).is_valid,file.name)
    def test_actual_geopdf_scale_and_corner_fit(self):
        with pymupdf.open(r.PDF) as doc:
            gs=[r.georeference(doc,doc[pn-1]) for pn,_,_ in r.LAYERS]
        for g in gs:
            self.assertLess(g['corner_residual_m'],.2)
            self.assertLess(max(abs(s-7.62) for s in g['scale_m_per_point']),.001)
            self.assertEqual(g,gs[0])

if __name__=='__main__':unittest.main(verbosity=2)
